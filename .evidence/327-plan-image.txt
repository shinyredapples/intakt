Issue #327, evidence E1: the job image of a PHP checkout, from a running install.
The build environment has no Docker daemon, so this is the plan the build sends to
POST /build, not the engine's build output. Provider notices at startup are left out.

Sample PHP checkout (php-sample/, one commit on main), composer.json:
{
  "name": "example/php-sample",
  "require": {"php": ">=8.4.1 <8.4.5", "ext-intl": "*", "ext-pdo_sqlite": "*"},
  "config": {"platform": {"php": "8.4.3"}}
}

$ bin/console intakt:job:plan-image php-sample; echo exit $?
Job image from composer.json (generated) for PHP 8.4.3 with intl, pdo_sqlite.

COPY --from=composer:2.11.1@sha256:8de2184c6de224f4689e8f80ed5f5edaf0030b6b8b7839a22a2bcb6722ac87d0 /usr/bin/composer /usr/local/bin/composer
COPY --from=mlocati/php-extension-installer:2.13.2 /usr/bin/install-php-extensions /usr/local/bin/install-php-extensions
RUN apt-get update \
    && apt-get install -y --no-install-recommends ca-certificates curl git unzip nodejs npm \
    && rm -rf /var/lib/apt/lists/*
RUN npm install --global @anthropic-ai/claude-code@2.1.322 @openai/codex@0.167.1 \
    && npm cache clean --force
RUN set -eu; for extension in intl pdo_sqlite; do \
        php -m | tr 'A-Z ' 'a-z-' | grep -qx "$extension" || install-php-extensions "$extension" || true; \
        php -m | tr 'A-Z ' 'a-z-' | grep -qx "$extension" || { echo "missing PHP extension: $extension" >&2; exit 1; }; \
    done
exit 0
